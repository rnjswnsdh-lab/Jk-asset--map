# 📥 접수 내역 수정·삭제 — Apps Script 서버 패치 (2026-11-28)

프런트(`hub.js`)는 이미 `request_update` · `request_delete` 액션을 호출합니다.
서버가 아직 이 액션을 모르면(`unknown_action`) 화면은 **이 브라우저에만 저장**하고
토스트로 `서버 반영은 Apps Script 패치 후 가능합니다`라고 알립니다.

아래 코드를 Apps Script 편집기(`Hub.gs`)에 붙이고 **새 버전으로 배포**하면
그때부터 목록에서 수정·삭제한 내용이 `lois` 시트에 실제로 반영됩니다.

## 권한 규칙 (확정)

| 계정 | 수정 | 삭제 |
| --- | --- | --- |
| 관리자(`role === 'admin'`) | 모든 접수 건 | 모든 접수 건 |
| 등록자 본인 | 자기가 올린 건 | 자기가 올린 건 |
| 그 외(내부 CM·물류TF·게스트·외부) | ✗ (버튼도 안 보임) | ✗ |

판정 기준은 **세션 사용자와 행의 등록자 일치**입니다. 우선순위:

1. `row.registrant_email` / `row.email` == 세션 이메일
2. `row.phone` 숫자만 비교 == 세션 전화번호
3. `row.name` + `row.company` 동시 일치

`name`/`phone`이 `비공개`로 마스킹된 행은 본인 확인에 쓰지 않습니다.

## 붙일 코드

```javascript
/* ── 접수 건 수정·삭제 (2026-11-28) — 등록자 본인 또는 관리자만 ── */
var REQUEST_EDITABLE = ['request_type', 'asset_name', 'region', 'area_py', 'price', 'rent',
                        'timeline', 'phone', 'email', 'status', 'memo'];

function reqActor_(session) {
  var me = (session && session.me) || {};
  return {
    admin: String(session && session.role) === 'admin',
    email: String(me.email || '').toLowerCase().trim(),
    phone: String(me.phone || '').replace(/[^0-9]/g, ''),
    name: String(me.name || '').trim(),
    company: String(me.company || '').trim()
  };
}

function reqIsOwner_(row, actor) {
  if (!row) return false;
  var email = String(row.registrant_email || row.email || '').toLowerCase().trim();
  if (actor.email && email && actor.email === email) return true;
  var phone = String(row.phone || '').replace(/[^0-9]/g, '');
  if (actor.phone && phone.length >= 7 && actor.phone === phone) return true;
  var nm = String(row.name || '').trim();
  var co = String(row.company || '').trim();
  if (actor.name && nm && nm !== '비공개' && actor.name === nm) {
    if (!co || !actor.company || co === actor.company) return true;
  }
  return false;
}

/** POST {action:'request_update', id, request:{…}} */
function requestUpdate_(body, session) {
  var actor = reqActor_(session);
  var sh = sheet_('lois');                       // 기존 헬퍼 이름에 맞춰 사용하세요
  var hit = findRowById_(sh, body.id);           // 헤더 'id' 컬럼 기준
  if (!hit) return { ok: false, error: 'not_found' };
  var row = hit.values;
  if (!actor.admin && !reqIsOwner_(row, actor)) return { ok: false, error: 'forbidden' };

  var patch = body.request || {};
  var cols = headerMap_(sh);
  REQUEST_EDITABLE.forEach(function (k) {
    if (patch[k] === undefined) return;
    if (k === 'memo' && !actor.admin && !reqIsOwner_(row, actor)) return;
    if (cols[k] == null) return;
    sh.getRange(hit.rowIndex, cols[k] + 1).setValue(String(patch[k]));
  });
  if (cols.updated_at != null) {
    sh.getRange(hit.rowIndex, cols.updated_at + 1).setValue(new Date());
  }
  logEvent_('request_update',
    body.id + ' · ' + (row.request_type || '') + ' · ' + (patch.asset_name || row.asset_name || '') +
    ' · ' + (actor.email || actor.name || session.role));
  return { ok: true, item: findRowById_(sh, body.id).values };
}

/** POST {action:'request_delete', id}  — 감사 기록을 남기기 위해 소프트 삭제를 권장합니다 */
function requestDelete_(body, session) {
  var actor = reqActor_(session);
  var sh = sheet_('lois');
  var hit = findRowById_(sh, body.id);
  if (!hit) return { ok: false, error: 'not_found' };
  var row = hit.values;
  if (!actor.admin && !reqIsOwner_(row, actor)) return { ok: false, error: 'forbidden' };

  var cols = headerMap_(sh);
  if (cols.status != null)   sh.getRange(hit.rowIndex, cols.status + 1).setValue('삭제');
  if (cols.deleted_at != null) sh.getRange(hit.rowIndex, cols.deleted_at + 1).setValue(new Date());
  if (cols.deleted_by != null) sh.getRange(hit.rowIndex, cols.deleted_by + 1).setValue(actor.email || actor.name || session.role);
  /* 완전 삭제를 원하면 위 3줄 대신: sh.deleteRow(hit.rowIndex); */
  logEvent_('request_delete',
    body.id + ' · ' + (row.request_type || '') + ' · ' + (row.asset_name || '') +
    ' · ' + (actor.email || actor.name || session.role));
  return { ok: true };
}
```

`doPost` 라우터에 두 줄을 추가합니다.

```javascript
if (action === 'request_update') return json_(requestUpdate_(body, session));
if (action === 'request_delete') return json_(requestDelete_(body, session));
```

## 함께 확인할 것

- `GET ?type=requests` 응답에 **행마다 `id`**(시트 고유 ID)가 실려야 화면에서 수정·삭제 대상이 잡힙니다.
  `id`가 없으면 `asset_id`를 대신 내려 주세요(프런트는 `id` → 없으면 첫 행에서 실패).
- 마스킹: 관리자·등록자 본인이 아닌 조회자에게는 지금처럼 `name`/`phone`/`email`을 `비공개`로 내려 주세요.
  수정·삭제 권한 판정은 **세션 기준**으로 서버가 결정합니다. (프런트 숨김은 보조 수단입니다)
- 삭제는 `deleted_at`·`status='삭제'`가 있으면 소프트 삭제로 남겨 감사 기록과 함께 보존합니다.
  목록 조회에서 `status === '삭제'` 행은 빼 주세요.
- 감사 로그(`log_event`)에 `request_update` / `request_delete`가 쌓이면
  🛰️ 관리자 → 접근·감사 로그 '전체 로그'에 `누가 · 언제 · 어떤 접수 건`이 그대로 보입니다.

## 배포 후 확인

1. 관리자로 로그인 → 홈 › 🧾 업무 신청 › 📥 접수 내역 보기 → 아무 행이나 눌러 상세 팝업.
2. `수정` → 물건명/상태 바꾸고 저장 → 목록에 즉시 반영, 새로고침해도 유지.
3. `삭제` → 확인 → 목록에서 사라지고 `lois` 시트에서도 `삭제` 표기(또는 행 제거).
4. 다른 계정(내부 CM·외부)으로 같은 행을 열어 **수정·삭제 버튼이 없는지** 확인.
